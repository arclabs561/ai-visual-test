Connect your application to Modbus
with only 3 lines of code:

modbus.cpp
MbusTcpMasterProtocol modbus;
modbus.openProtocol("192.168.0.120");
modbus.readMultipleRegisters(1, 40001, myData, sizeof(myData) / 2);
modbus.cs
var modbus = new MbusTcpMasterProtocol();
modbus.openProtocol("192.168.0.120");
modbus.readMultipleRegisters(1, 40001, myData);
modbus.vb
Dim modbus = new MbusTcpMasterProtocol();
modbus.openProtocol("192.168.0.120");
modbus.readMultipleRegisters(1, 40001, myData);

modbus.hostName := "192.168.0.120"; modbus.openProtocol;
modbus.readMultipleRegisters(1, 40001, myData);
